function actionCreateRoom(){
  const name = document.getElementById('hostName').value.trim();
  if(!name) return alert('Введите имя');
  const code = genCode();
  const room = {
    code, hostId state.playerId, phase 'lobby',
    players [{id state.playerId, name, alive true, traits {}, revealed {}}]
  };
  state.playerName = name;
  state.code = code;
  saveRoom(room);
  saveSession({code, playerId state.playerId, name});
  startPolling();
}

function actionJoinRoom(){
  const code = document.getElementById('joinCode').value.trim().toUpperCase();
  const name = document.getElementById('joinName').value.trim();
  if(!code  !name) return alert('Заполните поля');
  let room = getRoom(code);
  if(!room) return alert('Комната не найдена');
  if(room.players.some(p = p.name.toLowerCase() === name.toLowerCase())) return alert('Имя занято');
  room.players.push({id state.playerId, name, alive true, traits {}, revealed {}});
  state.playerName = name;
  state.code = code;
  saveRoom(room);
  saveSession({code, playerId state.playerId, name});
  startPolling();
}
